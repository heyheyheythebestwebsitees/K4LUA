local function rbUpdateDeflection()
    if not rbFighter or not rbFighter.Objects then return end
    for _, fighterObj in rbFighter.Objects do
        local player = fighterObj.Player
        if not player then continue end
        if not fighterObj.Entity or not fighterObj.Entity:IsAlive() or fighterObj:Get("IsSpectating") then
            rbDeflecting[player] = false; continue
        end
        local equipped = fighterObj.EquippedItem
        local isKatana = equipped and equipped.ViewModel and equipped.ViewModel.Name == "Katana"
        local isDeflecting = false
        if isKatana then isDeflecting = (equipped._attack_cooldown and equipped._attack_cooldown > tick()) or false end
        rbDeflecting[player] = isDeflecting
    end
end

local function rbIsEnemy(player)
    if player == rbLplr then return false end
    local duel = rbSpectate and rbSpectate.CurrentDuelSubject
    local localDueler = duel and duel:GetDueler(rbLplr)
    local localTeam = localDueler and localDueler:Get("TeamID") or nil
    if localTeam and duel and duel.Duelers then
        for _, dueler in duel.Duelers do
            if dueler.Player == player then
                local team = dueler:Get("TeamID")
                return team ~= localTeam
            end
        end
    end
    local pTeam = player:GetAttribute("TeamID")
    local lTeam = rbLplr:GetAttribute("TeamID")
    if pTeam and lTeam then return pTeam ~= lTeam end
    return true
end

local function rbGetClosestTarget()
    local char = rbLplr.Character
    if not char then return nil, nil, nil end
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil, nil, nil end
    local cP, cR, cH, cD = nil, nil, nil, 500
    for _, player in rbPlrs:GetPlayers() do
        if not rbIsEnemy(player) then continue end
        local pChar = player.Character
        if not pChar then continue end
        local pRoot = pChar:FindFirstChild("HumanoidRootPart")
        local pHead = pChar:FindFirstChild("Head")
        local pHum = pChar:FindFirstChildWhichIsA("Humanoid")
        if not (pRoot and pHead and pHum and pHum.Health > 0) then continue end
        local dist = (myRoot.Position - pRoot.Position).Magnitude
        if dist < cD then cD = dist; cP = player; cR = pRoot; cH = pHead end
    end
    return cP, cR, cH
end

local function rbHasKnifeViewModel(targetPlayer)
    if not targetPlayer then return false end
    local viewModels = rbWs:FindFirstChild("ViewModels")
    if not viewModels then return false end
    local targetName = targetPlayer.Name
    for _, model in viewModels:GetChildren() do
        if model:IsA("Model") and string.find(model.Name, targetName, 1, true) and string.find(model.Name, "Knife", 1, true) then return true end
    end
    return false
end

-- AdaptiveRage
local AdaptiveRage = {}
do
    local _conn, _stepConn, _charConn = nil, nil, nil
    local RENDER_NAME = "k4_AdaptiveRage_Restore"
    local CAMERA_NAME = "k4_AdaptiveRage_CamAnchor"
    local _realCF, _realChar = nil, nil
    local _voidCF = nil
    local _target = nil
    local _firing = false
    local _preParkCF = nil
    local _lastPark, _prevPark, _prevParkTgt = nil, nil, nil
    local _hiding, _primeUntil = true, 0
    local _hackTag, _lastSeen, _hackAccum = {}, {}, {}
    local _notified = nil
    local _meleeDwellStart = nil
    local PRIME_S, EYE_UP_SANE, MAX_DISTANCE, EYE_MUZZLE_SEP, RAGE_CLAMP_FRAC = 0.07, 2.5, 1200, 0.07, 0.30
    local _shootEnum, _useItem = nil, nil
    local _homieHoldUntil, _homieHoldCF = 0, nil
    local _fireFrames, _homieBursts = 0, 0
    local HOMIE_ARM_FRAMES, HOMIE_BURST_MAX = 8, 6

    local function isSanePos(p) return p == p and math.abs(p.X) < 1e5 and math.abs(p.Y) < 1e5 and math.abs(p.Z) < 1e5 end
    local function lookCF(fromPos, toPos)
        local dir = toPos - fromPos
        if dir.Magnitude < 1e-4 then dir = Vector3.new(0, -1, 0) end
        local up = Vector3.new(0, 1, 0)
        if math.abs(dir.Unit.Y) > 0.999 then up = Vector3.new(0, 0, 1) end
        return CFrame.lookAt(fromPos, fromPos + dir, up)
    end
    local function buildShotFields(cd, e, m, hp, aw, j, cf, jf)
        local os = hp.CFrame:PointToObjectSpace(aw)
        local hs = hp.Size * (cf or 0.45)
        os = Vector3.new(math.clamp(os.X, -hs.X, hs.X), math.clamp(os.Y, -hs.Y, hs.Y), math.clamp(os.Z, -hs.Z, hs.Z))
        if j then
            local jj = jf or 1.0
            os = Vector3.new(
                math.clamp(os.X + (math.random() - 0.5) * hs.X * jj, -hs.X, hs.X),
                math.clamp(os.Y + (math.random() - 0.5) * hs.Y * jj, -hs.Y, hs.Y),
                math.clamp(os.Z + (math.random() - 0.5) * hs.Z * jj, -hs.Z, hs.Z))
        end
        local hw = hp.CFrame:PointToWorldSpace(os)
        local osc = hp.CFrame:ToObjectSpace(CFrame.new(hw))
        cd[utf8.char(0)] = rbUtil:EncodeCFrame(e)
        cd[utf8.char(1)] = rbUtil:EncodeCFrame(m)
        cd[utf8.char(2)] = hp
        cd[utf8.char(3)] = rbUtil:EncodeCFrame(osc)
    end
    local function isProtected(player)
        if not player or not player.Character then return false end
        if player.Character:FindFirstChildOfClass("ForceField") then return true end
        return false
    end
    local function findTarget()
        local myChar = rbLplr.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local cands = {}
        for _, p in ipairs(rbPlrs:GetPlayers()) do
            if p ~= rbLplr and p.Character then
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hum and hum.Health > 0 and hrp then
                    local d = myRoot and (hrp.Position - myRoot.Position).Magnitude or 9999
                    local pTeam = p:GetAttribute("TeamID")
                    local lTeam = rbLplr:GetAttribute("TeamID")
                    local sameTeam = pTeam and lTeam and pTeam == lTeam
                    if not sameTeam and isSanePos(hrp.Position) and d <= MAX_DISTANCE then
                        table.insert(cands, { p=p, hp=hum.Health, d=d, prot=isProtected(p), hack=_hackTag[p]==true })
                    end
                end
            end
        end
        if #cands == 0 then return nil end
        table.sort(cands, function(a, b)
            if a.hack ~= b.hack then return a.hack end
            if a.prot ~= b.prot then return b.prot end
            if math.abs(a.hp - b.hp) > 10 then return a.hp < b.hp end
            return a.d < b.d
        end)
        return cands[1].p
    end
    local function voidAxis() local v = math.random(110000, 140000); return math.random(0, 1) == 0 and -v or v end
    local function rollVoid() return CFrame.new(voidAxis(), math.random(110000, 140000), voidAxis()) end
    local function restoreHome()
        local ch = rbLplr.Character
        local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
        if not hrp or not hrp.Parent then return end
        local back = _preParkCF
        _preParkCF = nil
        if back == nil then
            local real = _realCF
            if real and _realChar == ch then back = CFrame.new(real.Position) * hrp.CFrame.Rotation end
        end
        pcall(function() if back then hrp.CFrame = back end end)
    end
    local function displace(hrp, cf)
        if _preParkCF == nil then _preParkCF = hrp.CFrame end
        return pcall(function() hrp.CFrame = cf end)
    end
    local function weaponReady(it)
        if not it then return false end
        local okE, equipping = pcall(function() return it:IsEquipping() end)
        if okE and equipping then return false end
        if (it._reload_cooldown or 0) > tick() then return false end
        local isMelee = false
        pcall(function() isMelee = it.Info and it.Info.MaxAmmo == nil end)
        if isMelee then return true end
        local okA, ammo = pcall(function() return it:Get("Ammo") end)
        if not okA then return true end
        return type(ammo) == "number" and ammo > 0
    end
    local function adaptiveFire(eyePos, aimPos, hh, taps)
        local lf = rbFighter and rbFighter.LocalFighter
        local it = lf and lf.EquippedItem
        if not it then return 0 end
        pcall(function() it._shoot_cooldown = 0; it._shoot_cooldown_no_ammo = 0; it._last_shot = tick() - 1 end)
        if _shootEnum == nil then pcall(function() _shootEnum = rbEnum:ToEnum("StartShooting") end) end
        if _shootEnum == nil then return 0 end
        if _useItem == nil then pcall(function() _useItem = rbRepS.Remotes.Replication.Fighter.UseItem end) end
        if _useItem == nil then return 0 end
        local okId, objId = pcall(function() return it:Get("ObjectID") end)
        if not okId or not objId then return 0 end
        local eyeCF = lookCF(eyePos, aimPos)
        local muzzleCF = eyeCF - Vector3.new(0, EYE_MUZZLE_SEP, 0)
        local sent = 0
        pcall(function()
            for _ = 1, taps do
                local inner = {}
                buildShotFields(inner, eyeCF, muzzleCF, hh, aimPos, true, RAGE_CLAMP_FRAC, 1.0)
                local env = { [utf8.char(1)] = inner }
                _useItem:FireServer(objId, _shootEnum, env, nil)
                sent = sent + 1
            end
        end)
        return sent
    end
    local function hide(hrp, status)
        _firing = false; State.RageFiring = false; State.RageStatus = status
        _prevPark, _prevParkTgt, _lastPark = nil, nil, nil
        _hiding, _primeUntil = true, 0
        _meleeDwellStart = nil
        _homieHoldUntil, _homieHoldCF = 0, nil
        _fireFrames, _homieBursts = 0, 0
        displace(hrp, rollVoid())
    end
    local function adaptiveTick(ch, hrp)
        local tgt = _target
        if tgt and (not tgt.Parent or not tgt.Character or isProtected(tgt)) then tgt = nil end
        if not tgt then tgt = findTarget() end
        _target = tgt; State.RageTarget = tgt
        if not tgt or not tgt.Character then return hide(hrp, "No target") end
        local lf = rbFighter and rbFighter.LocalFighter
        local it = lf and lf.EquippedItem
        if not weaponReady(it) then return hide(hrp, "Reloading") end
        local hh = tgt.Character:FindFirstChild("HitboxHead") or tgt.Character:FindFirstChild("Head")
        if not hh then return hide(hrp, "Hiding") end
        if _hiding then _hiding = false; _primeUntil = tick() + PRIME_S + math.random() * 0.25 end
        local hpos = hh.Position
        if not isSanePos(hpos) then return hide(hrp, "Head voided") end
        _firing = true; State.RageFiring = true
        local park = hpos + Vector3.new(0, -1.5, 0)
        if not isSanePos(park) then return hide(hrp, "Hiding") end
        local wantHomie = getgenv().K4Adaptive and getgenv().K4Adaptive.HomiePulse and _fireFrames >= HOMIE_ARM_FRAMES and _homieBursts < HOMIE_BURST_MAX
        local desiredCF = CFrame.new(park)
        if wantHomie and _homieHoldCF == nil then
            local angle = math.random() * math.pi * 2
            local dist = 50000 + math.random() * 50000
            local homiePos = hrp.Position + Vector3.new(math.cos(angle) * dist, math.random(2000, 8000), math.sin(angle) * dist)
            _homieHoldCF = CFrame.new(homiePos); _homieHoldUntil = tick() + 0.30
            _homieBursts = _homieBursts + 1; _fireFrames = 0
        end
        if _homieHoldCF and tick() < _homieHoldUntil then desiredCF = _homieHoldCF; State.RageStatus = "Homie pulse"
        elseif _homieHoldCF then _homieHoldCF = nil; _homieHoldUntil = 0 end
        local parked = displace(hrp, desiredCF)
        if parked and _homieHoldCF == nil then _prevPark, _prevParkTgt, _lastPark = park, tgt, CFrame.new(park) end
        if _homieHoldCF then State.RageFiring = false; return end
        if tick() < _primeUntil then State.RageFiring = false; State.RageStatus = "Priming"; return end
        if isProtected(tgt) then State.RageStatus = "Protected"; return end
        local sendPos = park + Vector3.new(0, EYE_UP_SANE, 0)
        local sent = adaptiveFire(sendPos, hpos, hh, 1)
        State.RageStatus = sent > 0 and "Attacking" or "Missed"
        if sent > 0 then _fireFrames = _fireFrames + 1 end
        if _notified ~= tgt then _notified = tgt; pcall(function() if LHVisuals then LHVisuals.notifyTarget(tgt) end end) end
    end
    local function tickHackers(dt)
        if dt <= 0 or dt > 0.5 then return end
        for _, p in ipairs(rbPlrs:GetPlayers()) do
            if p ~= rbLplr and not _hackTag[p] then
                local c = p.Character
                local hrp = c and c:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local pos = hrp.Position
                    if not isSanePos(pos) then _hackTag[p] = true
                    else
                        local last = _lastSeen[p]
                        if last then
                            if (pos - last).Magnitude / dt > 120 then
                                _hackAccum[p] = (_hackAccum[p] or 0) + dt
                                if _hackAccum[p] >= 0.15 then _hackTag[p] = true end
                            else _hackAccum[p] = 0 end
                        end
                        _lastSeen[p] = pos
                    end
                end
            end
        end
    end
    function AdaptiveRage.start()
        if _conn then return end
        _firing = false; _notified = nil
        _homieHoldUntil, _homieHoldCF = 0, nil
        _fireFrames, _homieBursts = 0, 0
        pcall(function() R:UnbindFromRenderStep(RENDER_NAME) end)
        pcall(function() R:UnbindFromRenderStep(CAMERA_NAME) end)
        R:BindToRenderStep(CAMERA_NAME, Enum.RenderPriority.Camera.Value + 5, function()
            if not _preParkCF then return end
            local ch = rbLplr.Character
            local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
            if not hrp or not hrp.Parent then return end
            local delta = _preParkCF.Position - hrp.Position
            if isSanePos(delta) then Cam.CFrame = Cam.CFrame + delta end
        end)
        R:BindToRenderStep(RENDER_NAME, Enum.RenderPriority.First.Value - 1000, function() if not _firing then restoreHome() end end)
        _stepConn = R.Stepped:Connect(function() restoreHome() end)
        _charConn = rbLplr.CharacterAdded:Connect(function()
            _realCF, _realChar, _voidCF = nil, nil, nil
            _target, _notified = nil, nil
            _firing = false; _preParkCF = nil
            _prevPark, _prevParkTgt, _lastPark = nil, nil, nil
            _hiding, _primeUntil = true, 0
            _homieHoldUntil, _homieHoldCF = 0, nil
            _fireFrames, _homieBursts = 0, 0
        end)
        _conn = R.Heartbeat:Connect(function(dt)
            tickHackers(dt or 0)
            if not getgenv().Config.Enabled then AdaptiveRage.stop(); return end
            if getgenv().Config.Mode ~= "Adaptive" then AdaptiveRage.stop(); return end
            local ch = rbLplr.Character
            local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
            if not hrp or not hrp.Parent then return end
            if _preParkCF == nil and isSanePos(hrp.Position) then _realCF, _realChar = hrp.CFrame, ch end
            local hum = ch:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health <= 0 then
                _firing = false; State.RageFiring = false; State.RageStatus = "Dead"
                if not isSanePos(hrp.Position) then restoreHome() end
                return
            end
            adaptiveTick(ch, hrp)
        end)
    end
    function AdaptiveRage.stop()
        if _conn then _conn:Disconnect(); _conn = nil end
        if _stepConn then _stepConn:Disconnect(); _stepConn = nil end
        if _charConn then _charConn:Disconnect(); _charConn = nil end
        pcall(function() R:UnbindFromRenderStep(RENDER_NAME) end)
        pcall(function() R:UnbindFromRenderStep(CAMERA_NAME) end)
        _firing = false
        local hrp = rbLplr.Character and rbLplr.Character:FindFirstChild("HumanoidRootPart")

    local function easeInOut(a) return a * a * (3 - 2 * a) end

    local HOLO_SKEL_R6 = { {"Head","Torso"},{"Torso","Left Arm"},{"Torso","Right Arm"},{"Torso","Left Leg"},{"Torso","Right Leg"} }

    local function popSkeleton(char, dur, color)
        local vis = math.clamp(LHVisConfig.HoloVisibility, 0.2, 2)
        local tr0 = math.clamp(1 - 0.45 * vis, 0, 0.91)
        local hum = char:FindFirstChildOfClass("Humanoid")
        local isR6 = (hum and hum.RigType == Enum.HumanoidRigType.R6) or (char:FindFirstChild("Torso") ~= nil)
        local rig = isR6 and HOLO_SKEL_R6 or HOLO_SKEL_R15
        local model = Instance.new("Model"); model.Name = "_hs" .. math.random(10000, 99999)
        local parts = {}
        for _, pair in ipairs(rig) do
            local a, b = char:FindFirstChild(pair[1]), char:FindFirstChild(pair[2])
            if a and b then
                local ap, bp = a.Position, b.Position
                local len = (bp - ap).Magnitude
                if len > 0.05 and len < 20 then
                    local bone = Instance.new("Part")
                    bone.Shape = Enum.PartType.Cylinder
                    bone.Size = Vector3.new(len, 0.1 + 0.06 * vis, 0.1 + 0.06 * vis)
                    bone.Material = Enum.Material.Neon; bone.Color = color; bone.Transparency = tr0
                    bone.Anchored = true; bone.CanCollide = false; bone.CanQuery = false; bone.CastShadow = false; bone.Massless = true
                    bone.CFrame = CFrame.lookAt((ap + bp) * 0.5, bp) * CFrame.Angles(0, math.rad(90), 0)
                    bone.Parent = model; table.insert(parts, bone)
                end
            end
        end
        if #parts == 0 then model:Destroy(); return end
        model.Parent = getHoloFolder()
        local startT = tick()
        local conn
        conn = R.Heartbeat:Connect(function()
            if not model.Parent then if conn then conn:Disconnect() end return end
            local a = math.clamp((tick() - startT) / dur, 0, 1)
            local e = easeInOut(a)
            local tr = tr0 + (1 - tr0) * e
            for _, b in ipairs(parts) do b.Transparency = tr end
            if a >= 1 and conn then conn:Disconnect() end
        end)
        local Debris = game:GetService("Debris")
        Debris:AddItem(model, dur + 0.2)
    end

    local function createHologram(character, lethal)
        if not character or not character.Parent then return end
        local rp = character:FindFirstChild("HitboxHead") or character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("UpperTorso")
        if not rp then return end
        if (rp.Position - Cam.CFrame.Position).Magnitude > (LHVisConfig.HoloRange or 300) then return end
        if #getHoloFolder():GetChildren() >= 16 then return end
        local dur = math.clamp(LHVisConfig.HoloDuration or 3.5, 0.25, 10)
        local lethalOn = lethal and LHVisConfig.HoloLethal
        local mainColor = lethalOn and LHVisConfig.HoloLethalColor or LHVisConfig.HoloColor
        if LHVisConfig.HoloStyle == "Skeleton" then popSkeleton(character, dur, mainColor); return end
        local folder = getHoloFolder()
        local vis = math.clamp(LHVisConfig.HoloVisibility, 0.2, 2)
        local sc = 0.75 + 0.25 * vis
        local h0 = math.clamp(0.65 / vis, 0.1, 0.9)
        local function mkBall(size, transp)
            local b = Instance.new("Part")
            b.Shape = Enum.PartType.Ball; b.Size = Vector3.new(size, size, size)
            b.Material = Enum.Material.Neon; b.Color = mainColor; b.Transparency = transp
            b.Anchored = true; b.CanCollide = false; b.CanQuery = false; b.CastShadow = false; b.Massless = true
            return b
        end
        local core = mkBall(0.7 * sc, 0.05)
        local halo = mkBall(1.7 * sc, h0)
        local startCF = CFrame.new(rp.Position)
        core.CFrame = startCF; halo.CFrame = startCF
        core.Parent = folder; halo.Parent = folder
        local startT = tick()
        local conn
        conn = R.Heartbeat:Connect(function()
            if not core.Parent then if conn then conn:Disconnect() end return end
            local alpha = math.clamp((tick() - startT) / dur, 0, 1)
            local rise = 2.5 * (1 - (1 - alpha) * (1 - alpha))
            local cf = startCF + Vector3.new(0, rise, 0)
            core.CFrame = cf; halo.CFrame = cf
            core.Transparency = math.clamp(0.05 + 0.95 * alpha, 0, 1)
            halo.Transparency = math.clamp(h0 + (1 - h0) * alpha, 0, 1)
            if alpha >= 1 and conn then conn:Disconnect() end
        end)
        local Debris = game:GetService("Debris")
        Debris:AddItem(core, dur + 0.2); Debris:AddItem(halo, dur + 0.2)
    end

    function LHVisuals.notifyTarget(p)
        if not p or p == rbLplr or not p.Character then return end
        if not LHVisConfig.Holograms then return end
        local now = tick()
        if (now - (_hologramCooldowns[p] or 0)) < 0.35 then return end
        _hologramCooldowns[p] = now
        local hum = p.Character:FindFirstChildOfClass("Humanoid")
        local lethal = hum and hum.Health <= 0
        createHologram(p.Character, lethal)
    end

    function LHVisuals.enable()
        LHVisConfig.Enabled = true
        snapshotLighting()
        if Presets[LHVisConfig.Preset or "Neutral"] then Presets[LHVisConfig.Preset or "Neutral"]() end
        LHVisuals.reassertGrade(); LHVisuals.reassertBloom()
    end
    function LHVisuals.disable()
        LHVisConfig.Enabled = false
        if _gradeFx then pcall(function() _gradeFx:Destroy() end); _gradeFx = nil end
        if _bloomFx then pcall(function() _bloomFx:Destroy() end); _bloomFx = nil end
        restore()
    end

local function closest()
    if not root then return nil end
    local b, bd = nil, math.huge
    for _, p in ipairs(P:GetPlayers()) do
        if p ~= LP and p.Character then
            local h = p.Character:FindFirstChild("HumanoidRootPart")
            if h then
                local d = (root.Position - h.Position).Magnitude
                if d < bd then bd = d; b = p end
            end
        end
    end
    return b
end
local function getTgt(m)
    if m == "Closest" then return closest() end
    local l = {}
    for _, p in ipairs(P:GetPlayers()) do
        if p ~= LP and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then table.insert(l, p) end
    end
    if #l == 0 then return nil end
    if m == "Random" then return l[math.random(1, #l)] end
    if m == "Weakest" then
        table.sort(l, function(a, b)
            local ha = a.Character:FindFirstChildOfClass("Humanoid")
            local hb = b.Character:FindFirstChildOfClass("Humanoid")
            return (ha and ha.Health or 0) < (hb and hb.Health or 0)
        end)
        return l[1]
    end
    if m == "Strongest" then
        table.sort(l, function(a, b)
            local ha = a.Character:FindFirstChildOfClass("Humanoid")
            local hb = b.Character:FindFirstChildOfClass("Humanoid")
            return (ha and ha.Health or 0) > (hb and hb.Health or 0)
        end)
        return l[1]
    end
    return l[1]
end

local function computeRiotPos(targetRoot)
    local pos = targetRoot.Position
    local look = targetRoot.CFrame.LookVector
    local up = Vector3.new(0, 1, 0)
    local dist, height = byp.riot_bypass_distance or 3, byp.riot_bypass_height or 0
    local mode = byp.riot_bypass_position or "Behind"
    local base

    elseif mode == "Front" then base = pos + look * dist
    elseif mode == "Above" then base = pos + up * dist
    elseif mode == "Below" then base = pos - up * dist
    elseif mode == "Left" then base = pos - look:Cross(up).Unit * dist
    elseif mode == "Right" then base = pos + look:Cross(up).Unit * dist
    else base = pos - look * dist end
    return base + Vector3.new(0, height, 0)
end

        if not char then return end

        local spinCF = CFrame.Angles(0, spinAngle, 0)
        pcall(function()
            for _, v in pairs(char:GetDescendants()) do
                if v:IsA("Motor6D") then v.Transform = spinCF end
            end
        end)
    end)
    vst_conn = R.Heartbeat:Connect(function(dt)
        if not vst.enabled then return end
        pcall(function() workspace.FallenPartsDestroyHeight = -math.huge end)
        local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum or hum.Health <= 0 then return end

        if vtype ~= "Bait" then
            if hum:GetState() ~= Enum.HumanoidStateType.Physics then hum:ChangeState(Enum.HumanoidStateType.Physics) end
        end

        local function vst_do_defend_proximity_escape()
            if not vst.defend_proximity_escape then return end
            local escape_radius = math.max(vst.defend_proximity_radius or 20, 1)
            local escape_dist = math.max(vst.defend_escape_dist or 1000, 1)
            local my_pos = hrp.Position
            for _, p in ipairs(P:GetPlayers()) do
                if p ~= LP and p.Character then
                    local t_hrp = p.Character:FindFirstChild("HumanoidRootPart")
                    local t_hum = p.Character:FindFirstChildOfClass("Humanoid")
                    if t_hrp and t_hum and t_hum.Health > 0 then
                        if (my_pos - t_hrp.Position).Magnitude <= escape_radius then
                            local iters = math.clamp(vst.et_total or 1000, 1, 500)
                            local ir = math.max(vst.iter_radius or 500000, 1)
                            local iter_pos = my_pos
                            for _ = 1, iters do
                                local sx = (math.random() < 0.5) and 1 or -1
                                local sy = (math.random() < 0.5) and 1 or -1
                                local sz = (math.random() < 0.5) and 1 or -1
                                local ex = sx * (escape_dist + math.random() * escape_dist * 9)
                                local ey = sy * (ir * (0.5 + math.random() * 0.5))
                                local ez = sz * (escape_dist + math.random() * escape_dist * 9)
                                pcall(function()
                                    hrp.CFrame = CFrame.new(iter_pos.X + ex, iter_pos.Y + ey, iter_pos.Z + ez)
                                    hrp.AssemblyLinearVelocity = Vector3.zero
                                    hrp.AssemblyAngularVelocity = Vector3.zero
                                end)
                                iter_pos = hrp.Position
                            end
                            vst_last_pos = hrp.Position
                            return
                        end
                    end
                end
            end
        end

        local function vst_get_dir_offset(base_pos, dist)
            local dir_mode = vst.void_direction or "Random"
            local cam = Cam
            if dir_mode == "Forward" then
                local look = cam and Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z) or Vector3.new(0, 0, -1)
                if look.Magnitude > 0.001 then look = look.Unit else look = Vector3.new(0, 0, -1) end
                return Vector3.new(look.X * dist, (math.random() < 0.5 and 1 or -1) * dist, look.Z * dist)
            elseif dir_mode == "Backward" then
                local look = cam and Vector3.new(-cam.CFrame.LookVector.X, 0, -cam.CFrame.LookVector.Z) or Vector3.new(0, 0, 1)
                if look.Magnitude > 0.001 then look = look.Unit else look = Vector3.new(0, 0, 1) end
                return Vector3.new(look.X * dist, (math.random() < 0.5 and 1 or -1) * dist, look.Z * dist)
            elseif dir_mode == "Left" then
                local right = cam and cam.CFrame.RightVector or Vector3.new(1, 0, 0)
                local left = Vector3.new(-right.X, 0, -right.Z)
                if left.Magnitude > 0.001 then left = left.Unit else left = Vector3.new(-1, 0, 0) end
                return Vector3.new(left.X * dist, (math.random() < 0.5 and 1 or -1) * dist, left.Z * dist)
            elseif dir_mode == "Right" then
                local right = cam and Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z) or Vector3.new(1, 0, 0)
                if right.Magnitude > 0.001 then right = right.Unit else right = Vector3.new(1, 0, 0) end
                return Vector3.new(right.X * dist, (math.random() < 0.5 and 1 or -1) * dist, right.Z * dist)
            elseif dir_mode == "Up" then
                return Vector3.new((math.random() - 0.5) * dist * 0.2, dist, (math.random() - 0.5) * dist * 0.2)
            elseif dir_mode == "Down" then
                return Vector3.new((math.random() - 0.5) * dist * 0.2, -dist, (math.random() - 0.5) * dist * 0.2)
            elseif dir_mode == "Camera" then
                local look = cam and cam.CFrame.LookVector or Vector3.new(0, 0, -1)
                return look * dist
            elseif dir_mode == "Directional" then
                local xp, xn = vst.dir_xp or 5000, vst.dir_xn or 5000
                local yp, yn = vst.dir_yp or 5000, vst.dir_yn or 5000
                local zp, zn = vst.dir_zp or 5000, vst.dir_zn or 5000
                local rx = (math.random() < 0.5) and (math.random() * xp) or (-math.random() * xn)
                local ry = (math.random() < 0.5) and (math.random() * yp) or (-math.random() * yn)
                local rz = (math.random() < 0.5) and (math.random() * zp) or (-math.random() * zn)
                return Vector3.new(rx, ry, rz)
            else
                local sx = (math.random() < 0.5) and 1 or -1
                local sy = (math.random() < 0.5) and 1 or -1
                local sz = (math.random() < 0.5) and 1 or -1
                local ir2 = math.max(vst.iter_radius or 500000, 1)
                return Vector3.new(sx * (ir2 * (0.5 + math.random() * 0.5)), sy * (ir2 * (0.5 + math.random() * 0.5)), sz * (ir2 * (0.5 + math.random() * 0.5)))
            end
        end

        local function vst_do_defend_burst()
            vst_do_defend_proximity_escape()
            local iters = math.clamp(vst.et_total or 1000, 1, 500)
            local base_dist = math.max(vst.iter_radius or 500000, 1) * (0.5 + math.random() * 0.5)
            for i = 1, iters do
                local base = vst_last_pos or hrp.Position
                local offset = vst_get_dir_offset(base, base_dist)
                local target_cf = CFrame.new(base + offset)
                hrp.CFrame = target_cf
                vst_last_pos = hrp.Position
            end
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end

        local function vst_do_hide_burst()
            local base = hrp.Position
            local sx = (math.random() < 0.5) and 1 or -1
            local sz = (math.random() < 0.5) and 1 or -1
            pcall(function() hrp.CFrame = CFrame.new(base.X + sx * 1e6, base.Y - 9e25, base.Z + sz * 1e6) end)
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end

        if vtype == "Unhittability" then
            vst_do_defend_burst()
            local my_hrp_u = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
            if my_hrp_u then
                local y_range_u = math.max(vst.unhittability_y_range or 10, 1)
                pcall(function()
                    for _, p in pairs(P:GetPlayers()) do
                        if p ~= LP and p.Character then
                            local eh = p.Character:FindFirstChild("HumanoidRootPart")
                            if eh then
                                local y_diff = my_hrp_u.Position.Y - eh.Position.Y
                                if y_diff >= 0 and y_diff <= y_range_u then
                                    melee_fake_cf_store[p] = eh.CFrame
                                    local mt_u = getrawmetatable(eh)
                                    local old_ni_u = mt_u and rawget(mt_u, "__newindex")
                                    local fake_cf_u = CFrame.new(MELEE_FAKE_OFFSET)
                                    if old_ni_u then pcall(old_ni_u, eh, "CFrame", fake_cf_u) else rawset(eh, "CFrame", fake_cf_u) end
                                    eh.AssemblyLinearVelocity = Vector3.zero
                                else melee_fake_cf_store[p] = nil end
                            end
                        end
                    end
                end)
            end
        elseif vtype == "Defend" then vst_do_defend_burst()
        elseif vtype == "Attack" then
            local hide_dur = math.clamp(vst.hide_time or 0.3, 0.01, 1)
            local atk_dur = math.clamp(vst.attack_time or 0.3, 0.01, 1)
            local elapsed = tick() - vst_phase_start
            if vst_phase == "hide" then
                if vst.hide_mode == "Extreme" then vst_do_hide_burst()
                else vst_do_defend_burst() end
                if elapsed >= hide_dur then vst_phase = "attack"; vst_phase_start = tick() end
            else
                local target = closest()
                if target and target.Character then
                    local t_hrp = target.Character:FindFirstChild("HumanoidRootPart")
                    if t_hrp then
                        local head = target.Character:FindFirstChild("Head")
                        local iters = math.clamp(vst.et_total or 1000, 1, 500)
                        for _ = 1, iters do
                            local dx = (head or t_hrp).Position.X + (math.random()-0.5)*200
                            local dy = (head or t_hrp).Position.Y + (math.random()-0.5)*200
                            local dz = (head or t_hrp).Position.Z + (math.random()-0.5)*200
                            hrp.CFrame = CFrame.new(dx, dy, dz)
                            hrp.AssemblyLinearVelocity = Vector3.zero
                            hrp.AssemblyAngularVelocity = Vector3.zero
                        end
                        manipulation.fire_toward(head or t_hrp, target.Character)
                    end
                end
                if elapsed >= atk_dur then vst_phase = "hide"; vst_phase_start = tick() end
            end
        elseif vtype == "Bait" then
            local cur = hrp.Position
            hrp.CFrame = CFrame.new(cur.X, cur.Y - math.clamp(vst.bait_depth or 20, 1, 40), cur.Z)
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        elseif vtype == "Extreme" then
            local cur = hrp.Position
            pcall(function() hrp.CFrame = CFrame.new(cur.X, cur.Y - 9e25, cur.Z) end)
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        elseif vtype == "Random" then
            local iters = math.clamp(vst.et_total or 1000, 1, 500)
            local ir_rand = math.max(vst.iter_radius or 500000, 1)
            for i = 1, iters do
                local rbase = (vst_last_pos or hrp.Position)
                local offset = vst_get_dir_offset(rbase, ir_rand * (0.5 + math.random() * 0.5))
                pcall(function() hrp.CFrame = CFrame.new(rbase + offset) end)
                vst_last_pos = hrp.Position
            end
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

local vE, vP, iVP = 0, Vector3.new(), Vector3.new()

local function inVoid(p)
    local pr = RaycastParams.new(); pr.FilterType = Enum.RaycastFilterType.Include; pr.FilterDescendantsInstances = {workspace}
    return workspace:Spherecast(p, 1, Vector3.new(0, 0.01, 0), pr) ~= nil
end
local function tpE()
    local d = CFG.TP_RAND_ENABLED and (CFG.TP_RAND_MIN + math.random() * (CFG.TP_RAND_MAX - CFG.TP_RAND_MIN)) or 0
    if CFG.TP_NEG_BIAS and math.random() < 0.5 then d = -math.abs(d) end
    if not Cam or not CFG.TP_VIEWPORT_RAY then return Vector3.new(CFG.BASE_X + (math.random()*2-1)*math.abs(d), CFG.BASE_Y, CFG.BASE_Z + (math.random()*2-1)*math.abs(d)) end
    local vp = Cam.ViewportSize
    local ray = Cam:ViewportPointToRay(math.random()*vp.X, math.random()*vp.Y, math.abs(d))
    local b = ray.Origin
    local y = CFG.TP_PRESERVE_Y and (root and root.Position.Y or 0) or b.Y
    return Vector3.new(b.X, y, b.Z)
end
local function stepV(dt)
    local m = CFG.VOID_METHOD
    if m == "Drift" then vP = vP + Vector3.new(CFG.SPEED*dt, CFG.SPEED*dt*0.5, CFG.SPEED*dt*0.3); return vP
    elseif m == "Chaos" then vP = vP + Vector3.new((math.random()-0.5)*CFG.SPEED*dt*5, (math.random()-0.5)*CFG.SPEED*dt*5, (math.random()-0.5)*CFG.SPEED*dt*5); return vP
    elseif m == "Loop" then local r = math.min(CFG.RADIUS*0.8, 1e9 + (vE%100)*1e7); return Vector3.new(CFG.BASE_X + math.cos(vE*2)*r, (CFG.VOID_Y_MIN+CFG.VOID_Y_MAX)/2, CFG.BASE_Z + math.sin(vE*2)*r)
    elseif m == "Spiral" then local r = math.min(CFG.RADIUS*0.9, (vE%50)*2e9); return Vector3.new(CFG.BASE_X + math.cos(vE*3)*r, (CFG.VOID_Y_MIN+CFG.VOID_Y_MAX)/2, CFG.BASE_Z + math.sin(vE*3)*r)
    elseif m == "Quantum" then return Vector3.new(CFG.BASE_X + (math.random()*2-1)*CFG.RADIUS, (CFG.VOID_Y_MIN+CFG.VOID_Y_MAX)/2 + (math.random()*2-1)*(CFG.VOID_Y_MAX-CFG.VOID_Y_MIN)/2, CFG.BASE_Z + (math.random()*2-1)*CFG.RADIUS) end
    return tpE()
end
local function vKill()
    if not CFG.VOID_KILLER_ENABLED or not root then return end
    local mp = root.Position
    for _, p in ipairs(P:GetPlayers()) do
        if p ~= LP and p.Character then
            local h = p.Character:FindFirstChild("HumanoidRootPart")
            if h then
                local d = (h.Position - mp).Magnitude
                if d <= CFG.VOID_KILLER_RANGE and (inVoid(h.Position) or d < 500) then
                    if CFG.VOID_KILLER_SNAP_TO_MAP then
                        local rp = RaycastParams.new(); rp.FilterType = Enum.RaycastFilterType.Exclude; rp.FilterDescendantsInstances = {LP.Character, p.Character}
                        local rs = workspace:Raycast(Vector3.new(h.Position.X, h.Position.Y + 1000, h.Position.Z), Vector3.new(0, -5000, 0), rp)
                        if rs then pcall(function() h.CFrame = CFrame.new(rs.Position + Vector3.new(0, 3, 0)); h.AssemblyLinearVelocity = Vector3.zero end) end
                    end
                    if CFG.VOID_KILLER_FREEZE_TARGET then pcall(function() h.AssemblyLinearVelocity = Vector3.zero; h.AssemblyAngularVelocity = Vector3.zero end) end
                    if CFG.VOID_KILLER_NETWORK_OWN then pcall(function() h:SetNetworkOwner(LP) end) end
                    if CFG.VOID_KILLER_NO_CLIP then noClip(p.Character) end
                    if CFG.VOID_KILLER_AUTO_SHOOT then
                        if CFG.VOID_KILLER_AUTO_AIM then aim(root, h.Position) end
                        fire()
                    end
                end
            end
        end
    end
end
local function lockV(dt)
    local tp = stepV(dt)
    if CFG.VOID_ANTIVOID and inVoid(tp) then tp = tp + Vector3.new((math.random()-0.5)*200, math.random()*200, (math.random()-0.5)*200) end
    iVP = tp
    local h = root
    if not h then return end
    pcall(function() h.CFrame = CFrame.new(iVP); h.AssemblyLinearVelocity = Vector3.zero; h.AssemblyAngularVelocity = Vector3.zero end)
end

        if CFG.VOID_KILLER_ENABLED then vKill() end
    end)
end

local oA, oLE, oCR, oLS = 0, 0, 0, 0
local function startOrbit()
    killC("orbit")
    oA = 0; oLE = 0; oCR = cfg.orbitDist; oLS = 0
    if hum then pcall(function() hum:ChangeState(Enum.HumanoidStateType.Physics) end) end
    conns.orbit = R.Heartbeat:Connect(function(dt)
        if not cfg.orbitEnabled then killC("orbit"); return end
        if not root then return end
        if cfg.orbitNetworkOwn then pcall(function() root:SetNetworkOwner(LP) end) end
        if cfg.orbitNoClip and LP.Character then noClip(LP.Character) end
        if hum then pcall(function() if hum:GetState() ~= Enum.HumanoidStateType.Physics then hum:ChangeState(Enum.HumanoidStateType.Physics) end end) end
        local t = getTgt(cfg.orbitTarget)
        local tp = root.Position
        local th = nil
        if t and t.Character then
            th = t.Character:FindFirstChild("HumanoidRootPart")
            if th and (root.Position - th.Position).Magnitude <= cfg.orbitLockDist then
                tp = th.Position
                if cfg.orbitPredict then tp = tp + th.AssemblyLinearVelocity * cfg.orbitPredStr end
            end
        end
        local sp = cfg.orbitReverse and -cfg.orbitSpeed or cfg.orbitSpeed
        oA = (oA + math.rad(sp) * dt) % (2 * math.pi)
        oLE = oLE + dt
        local r, h = cfg.orbitDist, cfg.orbitHeight
        if cfg.orbitHeightOscillate then h = h + math.sin(oLE * cfg.orbitHeightOscSpeed * 2 * math.pi) * cfg.orbitHeightOscAmp end
        local x, y, z = 0, 0, 0
        local ax = cfg.orbitAxis
        if cfg.orbitMode == "Circle" then
            if ax == "XZ" then x = math.cos(oA)*r; z = math.sin(oA)*r; y = h
            elseif ax == "XY" then x = math.cos(oA)*r; y = math.sin(oA)*r; z = h
            elseif ax == "YZ" then y = math.cos(oA)*r; z = math.sin(oA)*r; x = h end
        elseif cfg.orbitMode == "Figure8" then x = math.sin(oA)*r; z = math.sin(oA)*math.cos(oA)*r*0.7; y = h
        elseif cfg.orbitMode == "Ellipse" then x = math.cos(oA)*r; z = math.sin(oA)*r*0.6; y = h
        elseif cfg.orbitMode == "SpiralIn" then oCR = math.max(1, oCR - dt * cfg.orbitDist * 0.2); x = math.cos(oA)*oCR; z = math.sin(oA)*oCR; y = h
        elseif cfg.orbitMode == "SpiralOut" then oCR = math.min(cfg.orbitDist*2, oCR + dt * cfg.orbitDist * 0.2); x = math.cos(oA)*oCR; z = math.sin(oA)*oCR; y = h end
        if cfg.orbitZigzag then x = x + math.sin(oLE * cfg.orbitZigzagFreq * 2 * math.pi) * cfg.orbitZigzagAmp end
        x = x + cfg.orbitOffX; z = z + cfg.orbitOffZ
        local dst = tp + Vector3.new(x, y, z)
        local al = math.min(1, 1 - (1 - cfg.orbitLerp)^(dt * 60))
        local sm = root.Position:Lerp(dst, al)
        pcall(function()
            if cfg.orbitFace then
                local cf = CFrame.new(sm, Vector3.new(tp.X, sm.Y, tp.Z))
                cf = cf * CFrame.Angles(math.rad(cfg.orbitPitchOff), math.rad(cfg.orbitYawOff), 0)
                root.CFrame = cf
            else root.CFrame = CFrame.new(sm) end
            if cfg.orbitVelZero then root.AssemblyLinearVelocity = Vector3.zero; root.AssemblyAngularVelocity = Vector3.zero end
        end)
        if cfg.orbitAutoShoot and th and t then
            local n = tick()
            if (root.Position - th.Position).Magnitude < cfg.orbitAttackRange and n - oLS > cfg.orbitShootCooldown then
                oLS = n
                if cfg.orbitAutoAim then aim(root, th.Position) end
                fire()
            end
        end
    end)
end
local function stopOrbit()
    killC("orbit"); cfg.orbitEnabled = false
    if hum then pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end) end
end

local lDT = 0
local pSC = {}
local function rayFire(e, m)
    if not e then return false end
    local o = e.Position + Vector3.new(0, 1.5, 0)
    local t = m + Vector3.new(0, 1.5, 0)
    local d = t - o
    local ds = d.Magnitude
    if ds < 0.01 then return false end
    local pr = RaycastParams.new(); pr.FilterType = Enum.RaycastFilterType.Exclude; pr.FilterDescendantsInstances = {LP.Character, e.Parent}
    return workspace:Raycast(o, d.Unit * ds, pr) == nil
end
local function animT(p)
    if not p or not p.Character then return false end
    local h = p.Character:FindFirstChildOfClass("Humanoid"); if not h then return false end
    local a = h:FindFirstChildOfClass("Animator"); if not a then return false end
    for _, t in ipairs(a:GetPlayingAnimationTracks()) do
        local n = (t.Animation and t.Animation.Name or ""):lower()
        if n:find("shoot") or n:find("fire") or n:find("attack") or n:find("swing") then return true end
    end
    return false
end
local function perp(f, m)
    local fl = Vector3.new(f.X - m.X, 0, f.Z - m.Z)
    if fl.Magnitude < 0.01 then fl = Vector3.new(1, 0, 0) end
    fl = fl.Unit
    local p1 = Vector3.new(-fl.Z, 0, fl.X)
    local p2 = Vector3.new(fl.Z, 0, -fl.X)
    return (math.random(0, 1) == 0 and p1 or p2) + Vector3.new(0, (math.random() - 0.5) * 0.4, 0)
end
local function startPred()

    conns.pred = R.Heartbeat:Connect(function(dt)
        if not dg.predEnabled then killC("pred"); return end
        if not root or (hum and hum.Health <= 0) then return end

                    end

        if not th then return end

        local f = pSC[th] and (n - pSC[th]) < dg.predReactTime
        local aT = false
        do
            local o = aimOrig(th)
            if o then
                local tm = mp - o.Position
                if tm.Magnitude > 0.01 and o.CFrame.LookVector:Dot(tm.Unit) > dg.predShootDot then aT = true end
            end
        end
        local lT = dg.predRaycast and rayFire(thH, mp)
        local anT = dg.predAnimDetect and animT(th)
        if not (f or aT or lT or anT) then return end
        lDT = n
        local d = perp(thH.Position, mp)
        local np = mp + d * dg.predDodgeDist
        for i = 1, 3 do pcall(function() root.CFrame = CFrame.new(np + Vector3.new((math.random()-0.5)*4, 0, (math.random()-0.5)*4)); root.AssemblyLinearVelocity = d * 150 end); R.Heartbeat:Wait() end
        if dg.predNotifyAnywhere and (thH.Position - mp).Magnitude < 500 then Notify("dodge: " .. th.Name, 1.5) end
    end)
end
local function stopPred() killC("pred"); dg.predEnabled = false end

local abD, lAB, gCD = {}, 0, {}
local function dodgeDir(t)
    if not root then return Vector3.new(1, 0, 0) end
    local fl = Vector3.new(t.X - root.Position.X, 0, t.Z - root.Position.Z)
    local aw = fl.Magnitude > 0.01 and -fl.Unit or Vector3.new(1, 0, 0)
    if ab.dodgeMode == "Away" then return aw end
    if ab.dodgeMode == "Perpendicular" then
        local p = Vector3.new(-aw.Z, 0, aw.X)
        return math.random(0, 1) == 0 and p or -p
    end
    local a = math.random() * 2 * math.pi
    return Vector3.new(math.cos(a), 0, math.sin(a))
end
local function doDodge(p, r)
    if not root then return end
    local n = tick()
    if n - lAB < ab.dodgeCooldown then return end
    lAB = n
    local tp = Vector3.zero
    if p and p.Character then local h = p.Character:FindFirstChild("HumanoidRootPart"); if h then tp = h.Position end end
    local d = dodgeDir(tp)
    for i = 1, 3 do pcall(function() root.CFrame = root.CFrame + d * (ab.velDodgeDist * 0.005); root.AssemblyLinearVelocity = d * 180 end); R.Heartbeat:Wait() end
    if ab.notifyOnDodge then Notify("dodge: " .. (p and p.Name or "?") .. " [" .. r .. "]", 1.5) end
end
local function startAB()
    killC("antiBait"); abD = {}
    for _, p in ipairs(P:GetPlayers()) do if p ~= LP then abD[p] = {lastPos=nil, tpTimes={}} end end
    conns.antiBait = R.Heartbeat:Connect(function(dt)
        if not ab.enabled then killC("antiBait"); return end
        if not root then return end
        if ab.antiBaitNoClip and LP.Character then noClip(LP.Character) end
        if ab.antiBaitNetworkOwn then netOwn(root) end
        local mp = root.Position
        local n = tick()
        local rad = ab.infiniteRange and ab.infiniteRadius or 500
        for _, p in ipairs(P:GetPlayers()) do
            if p ~= LP and p.Character then
                local h = p.Character:FindFirstChild("HumanoidRootPart")
                if h then
                    local d = abD[p]; if not d then abD[p] = {lastPos=nil, tpTimes={}}; d = abD[p] end
                    local cp, cv = h.Position, h.AssemblyLinearVelocity
                    if ab.gunPredEnabled then
                        local o = aimOrig(p) or h
                        local tm = mp - o.Position
                        if tm.Magnitude > 0.01 and tm.Magnitude < rad and o.CFrame.LookVector:Dot(tm.Unit) > ab.gunPredDot then
                            local k = p.UserId
                            if not gCD[k] or n - gCD[k] > ab.gunPredCooldown then gCD[k] = n; doDodge(p, "gun") end
                        end
                    end
                    if ab.baitHookActivate and pSC[p] and (n - pSC[p]) < ab.baitReactionWindow then doDodge(p, "fire") end
                    if ab.dodgeOnShoot then
                        local lk = h.CFrame.LookVector
                        local tm = mp - h.Position
                        if tm.Magnitude > 0.01 and lk:Dot(tm.Unit) > ab.shootLookThresh then doDodge(p, "shoot") end
                    end
                    if ab.velSpikeEnabled and cv.Magnitude > ab.velThreshold then
                        local tm = mp - cp
                        if tm.Magnitude > 0.01 and cv.Unit:Dot(tm.Unit) > 0.6 then doDodge(p, "vel") end
                    end
                    if ab.flickerEnabled and d.lastPos then
                        if (cp - d.lastPos).Magnitude > ab.flickerTpDist then d.tpTimes[#d.tpTimes+1] = tick() end
                        local cl = {}
                        for _, t in ipairs(d.tpTimes) do if n - t < ab.flickerWindow then cl[#cl+1] = t end end
                        d.tpTimes = cl
                        if #d.tpTimes >= ab.flickerMinTps then d.tpTimes = {}; doDodge(p, "flicker") end
                    end
                    d.lastPos = cp
                end
            end
        end
    end)
end
local function stopAB() killC("antiBait"); ab.enabled = false; abD = {}; gCD = {} end

local reHB, reOR = nil, nil
local function reHbox(p)
    if not p.Character or not re.expandHitboxes then return end
    local s = re.hitboxSize
    for _, x in ipairs(p.Character:GetDescendants()) do
        if x:IsA("BasePart") then
            if not re.originalParts[x] then re.originalParts[x] = {size=x.Size, transparency=x.Transparency, cancollide=x.CanCollide} end
            pcall(function() x.Size = Vector3.new(s, s, s); x.Transparency = re.hitboxTransparency; x.CanCollide = false; x.Massless = true end)
        end
    end
end
local function reClr(p)
    if not p.Character then return end
    for _, x in ipairs(p.Character:GetDescendants()) do
        if x:IsA("BasePart") then
            local o = re.originalParts[x]
            if o then pcall(function() x.Size = o.size; x.Transparency = o.transparency; x.CanCollide = o.cancollide; x.Massless = false end) end
        end
    end
end
local function reRC()
    if not re.patchRaycast or reOR or not hookfunction then return end
    pcall(function()
        reOR = workspace.Raycast
        local o = reOR
        hookfunction(o, function(s, o2, d, p2)
            if re.enabled then return o(s, o2, d.Unit * re.range, p2) end
            return o(s, o2, d, p2)
        end)
    end)
end

    re.enabled = true; reRC()
    for _, p in ipairs(P:GetPlayers()) do if p ~= LP and (not re.teamCheck or p.Team ~= LP.Team) then reHbox(p) end end

    reHB = R.Heartbeat:Connect(function()

        if re.rangeExtNetworkOwn and root then netOwn(root) end

        for _, p in ipairs(P:GetPlayers()) do
            if c >= re.maxEnemies then break end
            if p ~= LP and (not re.teamCheck or p.Team ~= LP.Team) then reHbox(p); c = c + 1 end

end

local tC, tPR, tRC, tAcc = nil, nil, nil, 0
local function tPos(rf)
    local d = tr.offsetDist
    local j = tr.jitter and tr.jitterAmp or 0
    local b = rf.Position
    local ox, oy, oz = 0, 0, 0
    local m = tr.offsetMode
    if m == "Ahead" then local l = rf.LookVector; ox = l.X * d; oz = l.Z * d
    elseif m == "Behind" then local l = rf.LookVector; ox = -l.X * d; oz = -l.Z * d
    elseif m == "Perpendicular" then local r = rf.RightVector; local s = math.random() < 0.5 and 1 or -1; ox = r.X * d * s; oz = r.Z * d * s
    elseif m == "Above" then oy = math.abs(d)
    elseif m == "Below" then oy = -math.abs(d) end
    oy = oy + tr.offsetY
    ox = ox + (math.random()-0.5)*j
    oz = oz + (math.random()-0.5)*j
    return CFrame.new(b.X + ox, b.Y + oy, b.Z + oz)
end
local function startTR()
    if tC then tC:Disconnect() end
    if tPR then tPR:Disconnect() end

        if tr.enabled and root and tRC and tr.snapBack then root.CFrame = tRC; if tr.translocVelZero then root.AssemblyLinearVelocity = Vector3.zero end end
    end)
    tC = R.Heartbeat:Connect(function(dt)
        if not tr.enabled then stopTR(); return end
        if not root then return end
        if tr.translocNoClip and LP.Character then noClip(LP.Character) end
        tRC = root.CFrame
        tAcc = tAcc + dt
        local iv = 1 / math.max(tr.frequency, 1)
        if tAcc < iv then return end
        tAcc = tAcc % iv
        for i = 1, math.max(tr.layers, 1) do
            local fcf = tPos(tRC)
            local fv = tr.fakeVel and Vector3.new((math.random()-0.5)*tr.fakeVelMag*2, (math.random()-0.5)*tr.fakeVelMag*2, (math.random()-0.5)*tr.fakeVelMag*2) or Vector3.zero
            pcall(function() root.CFrame = fcf; root.AssemblyLinearVelocity = fv end)
        end
    end)
end
local function stopTR()

local aAA, aAJ, aART, aARM = 0, 0, 0, 1
local aMT, aRA, aBA, aCT, aLT = 0, 0, 0, 0, 0
local function startAA()
    killC("antiAim")
    conns.antiAim = R.Heartbeat:Connect(function(dt)
        if not aa.enabled then killC("antiAim"); return end
        if not root then return end
        if aa.aaNoClip and LP.Character then noClip(LP.Character) end
        if aa.aaNetworkOwn then netOwn(root) end
        aMT = aMT + dt; aBA = aBA + dt; aRA = aRA + dt; aCT = aCT + dt; aLT = aLT + dt
        local sp = aa.speed
        if aa.randomSpeed then aART = aART + dt; if aART > 0.1 then aART = 0; aARM = 0.2 + math.random() * 0.8 end; sp = sp * aARM end
        local eY, eP, eR = 0, 0, 0
        if aa.breatheEffect then eP = eP + math.sin(aBA * aa.breatheSpeed * 2 * math.pi) * aa.breatheAmp end
        if aa.microJitter and aMT >= (1 / math.max(aa.microJitterSpeed, 1)) then
            aMT = 0
            eY = eY + (math.random() - 0.5) * aa.microJitterAmp * 2
            eP = eP + (math.random() - 0.5) * aa.microJitterAmp
        end
        if aa.desyncMode then eY = eY + aa.desyncOffset end
        if aa.randomFlip and math.random() < aa.flipChance then eY = eY + 180 end
        if aa.enhancedMode then
            eY = eY + (math.random() - 0.5) * 90
            eP = eP + (math.random() - 0.5) * 90
            if aa.chaosMode and aCT >= aa.chaosInterval then
                aCT = 0
                eY = eY + (math.random() - 0.5) * 180
                eP = eP + (math.random() - 0.5) * 180
                eR = eR + (math.random() - 0.5) * 180
            end
            if aa.lookAtRandom and aLT >= aa.lookAtRandomInterval then aLT = 0; eY = eY + math.random() * 360; eP = eP + (math.random() - 0.5) * 180 end
        end
        local rA = 0
        if aa.rollEnabled then rA = (aRA * aa.rollSpeed * math.pi / 180) % (2 * math.pi) end
        if aa.mode == "Spin" then aAA = aAA + sp * dt; root.CFrame = CFrame.new(root.Position) * CFrame.Angles(math.rad(eP), math.rad(aAA + eY), rA + eR)
        elseif aa.mode == "Jitter" then
            local p = aa.jitterPitch and aa.pitchAngle or 0
            local y = aa.jitterYaw and aa.yawAngle or 0
            root.CFrame = root.CFrame * CFrame.Angles(math.rad((aAJ == 0 and p or -p) + eP), math.rad((aAJ == 0 and y or -y) + eY), rA + eR)
            aAJ = (aAJ + 1) % 2
        elseif aa.mode == "Static" then root.CFrame = CFrame.new(root.Position) * CFrame.Angles(math.rad(aa.staticPitch + eP), math.rad(aa.staticYaw + eY), rA + eR)
        elseif aa.mode == "Up" then root.CFrame = CFrame.new(root.Position) * CFrame.Angles(math.rad(90 + eP), math.rad(eY), rA + eR)
        elseif aa.mode == "Down" then root.CFrame = CFrame.new(root.Position) * CFrame.Angles(math.rad(-90 + eP), math.rad(eY), rA + eR)
        elseif aa.mode == "RandomSpin" then aAA = aAA + sp * dt * (math.random() * 2 - 1); root.CFrame = CFrame.new(root.Position) * CFrame.Angles(math.rad(aa.customPitch + eP), math.rad(aAA + eY), rA + eR)
        elseif aa.mode == "Custom" then root.CFrame = CFrame.new(root.Position) * CFrame.Angles(math.rad(aa.customPitch + eP), math.rad(aa.customYaw + eY), rA + eR) end
    end)

local function espStep()
    if not esp.enabled then return end
    local cam = workspace.CurrentCamera
    if not cam then return end
    local vp = cam.ViewportSize
    for _, p in ipairs(P:GetPlayers()) do
        if p == LP or not p.Character then espHide(espCache[p])
        elseif esp.teamCheck and p.Team == LP.Team then espHide(espCache[p])
        else
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            local head = p.Character:FindFirstChild("Head")
            local h = p.Character:FindFirstChildOfClass("Humanoid")
            if not hrp or not head or not h then espHide(espCache[p])
            else
                local dist = (cam.CFrame.Position - hrp.Position).Magnitude
                if dist > esp.maxDist then espHide(espCache[p])
                else
                    local d = espGet(p)
                    local headPos, onH = cam:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                    local footPos, onF = cam:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                    if not onH and not onF then espHide(d)
                    else
                        local boxH = math.abs(footPos.Y - headPos.Y)
                        local boxW = boxH * 0.55
                        local bx = headPos.X - boxW / 2
                        local by = headPos.Y
                        if esp.box then
                            d.box.Visible = true
                            d.box.Position = UDim2.fromOffset(bx, by)
                            d.box.Size = UDim2.fromOffset(boxW, boxH)
                            d.boxStroke.Color = esp.boxColor
                            d.boxStroke.Transparency = 1 - esp.boxTrans
                        else d.box.Visible = false end
                        if esp.fillBox then
                            d.fill.Visible = true
                            d.fill.Position = UDim2.fromOffset(bx, by)
                            d.fill.Size = UDim2.fromOffset(boxW, boxH)
                            d.fill.BackgroundColor3 = esp.fillColor
                            d.fill.BackgroundTransparency = 1 - esp.fillTrans
                        else d.fill.Visible = false end
                        if esp.name then
                            d.name.Visible = true; d.name.Text = p.Name; d.name.TextSize = esp.textSize; d.name.TextColor3 = esp.textColor
                            d.name.Position = UDim2.fromOffset(bx + boxW / 2 - 100, by - 18)
                            d.name.Size = UDim2.fromOffset(200, 16)
                        else d.name.Visible = false end
                        if esp.dist then
                            d.dist.Visible = true; d.dist.Text = "[" .. math.floor(dist) .. "]"; d.dist.TextSize = esp.textSize - 2; d.dist.TextColor3 = esp.textColor
                            d.dist.Position = UDim2.fromOffset(bx + boxW / 2 - 100, by + boxH + 2)
                            d.dist.Size = UDim2.fromOffset(200, 14)
                        else d.dist.Visible = false end
                        if esp.health then
                            local hpc = math.clamp(h.Health / h.MaxHealth, 0, 1)
                            d.hpBg.Visible = true; d.hpBg.Position = UDim2.fromOffset(bx - 6, by); d.hpBg.Size = UDim2.fromOffset(3, boxH)
                            d.hp.Visible = true; d.hp.Position = UDim2.fromOffset(0, boxH - boxH * hpc); d.hp.Size = UDim2.fromOffset(3, boxH * hpc)
                            d.hp.BackgroundColor3 = Color3.fromRGB(math.floor(255 * (1 - hpc)), math.floor(255 * hpc), 0)
                        else d.hpBg.Visible = false; d.hp.Visible = false end
                        if esp.tracer then
                            d.tracer.Visible = true; d.tracer.BackgroundColor3 = esp.tracerColor
                            local fromX, fromY = vp.X / 2, vp.Y
                            local toX, toY = headPos.X, by + boxH
                            local dx, dy = toX - fromX, toY - fromY
                            local len = math.sqrt(dx * dx + dy * dy)
                            local ang = math.atan2(dy, dx)
                            d.tracer.AnchorPoint = Vector2.new(0, 0.5)
                            d.tracer.Position = UDim2.fromOffset(fromX, fromY)
                            d.tracer.Size = UDim2.fromOffset(len, 1)
                            d.tracer.Rotation = math.deg(ang)
                        else d.tracer.Visible = false end
                    end
                end
            end
        end
    end
end

local function getFraudFPS()
    local base = tonumber(spoofer.fps_value) or 60
    if spoofer_state.last_base_fps == nil then spoofer_state.last_base_fps = base end
    local variation = math.random(-8, 12)
    spoofer_state.fps_trend = math.clamp(spoofer_state.fps_trend + math.random(-3, 5), -15, 15)
    local tv = variation + spoofer_state.fps_trend
    if math.random(1, 100) <= 15 then tv = tv + math.random(-10, 20) end
    if math.random(1, 100) <= 8 then tv = tv - math.random(5, 15) end
    local nv = base + tv
    if math.random(1, 100) <= 5 then nv = nv + math.random(20, 45) end
    if nv < 10 then nv = math.random(10, 25) end
    if nv > 144 then nv = math.random(120, 144) end
    if math.random(1, 100) <= 3 then nv = math.random(30, 50) end
    spoofer_state.last_base_fps = nv
    return tostring(math.floor(nv))
end
local function getFraudMS()
    local base = tonumber(spoofer.ms_value) or 50
    if spoofer_state.last_base_ms == nil then spoofer_state.last_base_ms = base end
    spoofer_state.ms_trend = math.clamp(spoofer_state.ms_trend + math.random(-4, 6), -20, 25)
    local spike = 0
    if math.random(1, 100) <= 12 then spike = math.random(15, 45) end
    if math.random(1, 100) <= 7 then spike = spike - math.random(10, 30) end
    local variation = math.random(-12, 18)
    local nv = base + variation + spoofer_state.ms_trend + spike
    if nv < 8 then nv = math.random(8, 20) end
    if nv > 250 then nv = math.random(180, 250) end
    if math.random(1, 100) <= 10 then nv = nv + math.random(5, 25) end
    if math.random(1, 100) <= 6 then nv = math.random(100, 200) end
    spoofer_state.last_base_ms = nv
    return tostring(math.floor(nv))
end
local function applyFPSSpoof()
    for i = 1, #spoofer_state.fps_conns do spoofer_state.fps_conns[i]:Disconnect() end
    spoofer_state.fps_conns = {}; spoofer_state.fps_fraud_running = false; spoofer_state.last_base_fps = nil; spoofer_state.fps_trend = 0
    if not spoofer.fps_enabled then return end
    if spoofer.fps_fraud then
        spoofer_state.fps_fraud_running = true
        task.spawn(function()
            while spoofer_state.fps_fraud_running do
                local target = getFraudFPS()
                for i = 1, #spoofer_state.labels do
                    local label = spoofer_state.labels[i]
                    local spoofed = label.Text:gsub("%d+fps", target .. "fps")
                    if spoofed ~= label.Text then label.Text = spoofed end
                end
                task.wait(0.95 + math.random() * 0.4)
            end
        end)
    else
        local target = spoofer.fps_value
        for i = 1, #spoofer_state.labels do
            local label = spoofer_state.labels[i]
            spoofer_state.fps_conns[#spoofer_state.fps_conns + 1] = label:GetPropertyChangedSignal("Text"):Connect(function()
                local spoofed = label.Text:gsub("%d+fps", target .. "fps")
                if spoofed ~= label.Text then label.Text = spoofed end
            end)
        end
    end
end

local function xLead(c, from)
    if not c then return Vector3.new() end
    local hum = c:FindFirstChildOfClass("Humanoid")
    local h = c:FindFirstChild("HumanoidRootPart")
    if not hum or not h then return Vector3.new() end
    local lat = (LP:GetNetworkPing() or 0.05) + ((XC.ServerProcessingMs or 30) / 1000)
    local lead = Vector3.new()
    if XC.PredictiveLead then
        local ply = P:GetPlayerFromCharacter(c)
        local tv = h.AssemblyLinearVelocity
        local cv = ply and _xVel[ply]
        if cv then if (tv - cv).Magnitude > 25 then tv = cv end
        else if tv.Magnitude > 100 then tv = Vector3.new() end end
        if tv.Magnitude > 120 then tv = Vector3.new() end
        if hum.MoveDirection.Magnitude > 0.1 then lead = lead + Vector3.new(tv.X, 0, tv.Z) * lat end
        lead = lead + Vector3.new(0, tv.Y * lat, 0)
    end
    local cap = XC.LeadCap or 15
    if lead.Magnitude > cap then lead = lead.Unit * cap end
    return lead
end

    local sb = o.stickyBonus or 0
    local vp = Cam.ViewportSize
    local cc = Vector2.new(vp.X * 0.5, vp.Y * 0.5)
    local b, bp, bs = nil, nil, math.huge
    for _, p in ipairs(P:GetPlayers()) do
        if p ~= LP and xValid(p, false) then
            local pt = xPick(p.Character, mode)
            if pt and (not cv or xVis(pt.Position)) then
                local sp, on = Cam:WorldToViewportPoint(pt.Position)
                if on and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - cc).Magnitude
                    if d <= fov then
                        local s = d
                        if p == sk then s = s * (1 - sb) end
                        if s < bs then bs, b, bp = s, p, pt end
                    end
                end
            end
        end
    end
    return b, bp
end
local function xLook(f, t)
    local d = t - f
    if d.Magnitude < 1e-4 then d = Vector3.new(0, -1, 0) end
    local up = Vector3.new(0, 1, 0)
    if math.abs(d.Unit.Y) > 0.999 then up = Vector3.new(0, 0, 1) end
    return CFrame.lookAt(f, f + d, up)
end
local function xBuild(cd, e, m, hp, aw, j, cf, jf)
    local os = hp.CFrame:PointToObjectSpace(aw)
    local hs = hp.Size * (cf or 0.45)
    os = Vector3.new(math.clamp(os.X, -hs.X, hs.X), math.clamp(os.Y, -hs.Y, hs.Y), math.clamp(os.Z, -hs.Z, hs.Z))
    if j then
        local jj = jf or 1
        os = Vector3.new(
            math.clamp(os.X + (math.random() - 0.5) * hs.X * jj, -hs.X, hs.X),
            math.clamp(os.Y + (math.random() - 0.5) * hs.Y * jj, -hs.Y, hs.Y),
            math.clamp(os.Z + (math.random() - 0.5) * hs.Z * jj, -hs.Z, hs.Z))
    end
    local hw = hp.CFrame:PointToWorldSpace(os)
    local osc = hp.CFrame:ToObjectSpace(CFrame.new(hw))
    cd[utf8.char(0)] = rbUtil:EncodeCFrame(e)
    cd[utf8.char(1)] = rbUtil:EncodeCFrame(m)
    cd[utf8.char(2)] = hp
    cd[utf8.char(3)] = rbUtil:EncodeCFrame(osc)
end
local function xEnc(cd, hp, tc, cp, co)
    if not hp or not cd or not rbUtil then return false end
    local lead = xLead(tc, cp)
    local off = (typeof(co) == "Vector3") and co or Vector3.zero
    local lp = hp.Position + lead + off
    local e = xLook(cp, lp)
    local mp = cp + (e.RightVector * 0.2) + Vector3.new(0, -0.07, 0)
    local m = xLook(mp, lp)
    xBuild(cd, e, m, hp, lp, XC.SilentAimJitter, 0.45, 1.0)
    return true
end
local function xSVis(p)
    if not p or not p:IsA("BasePart") then return false end
    if not XC.SilentAimMultipoint then return xVis(p.Position) end
    if xVis(p.Position) then return true end
    local n = math.max(1, math.floor(XC.SilentAimMultipointCount or 5))
    local r = math.min(p.Size.X, p.Size.Y, p.Size.Z) * 0.4
    for i = 1, n do
        local a = (i / n) * math.pi * 2
        local off = Vector3.new(math.cos(a) * r, ((i % 2 == 0) and 0.5 or -0.5) * r, math.sin(a) * r)
        if xVis(p.Position + off) then return true end
    end
    return false
end
local function xSP(c, pr)
    if not c then return pr end
    if xSVis(pr) then return pr end
    for _, n in ipairs(X_HEAD) do local p = c:FindFirstChild(n); if p and p ~= pr and xSVis(p) then return p end end
    if XC.SilentAimTorsoFallback then for _, n in ipairs(X_TORSO) do local p = c:FindFirstChild(n); if p and xSVis(p) then return p end end end
    return pr
end
local function xHook(tbl, m, key)
    if type(tbl) ~= "table" then return end
    local o = rawget(tbl, m)
    if type(o) ~= "function" then return end
    if shared[key] then return end
    shared[key] = o
    if setreadonly then pcall(setreadonly, tbl, false) end
    tbl[m] = function(self, ...)
        local r = { o(self, ...) }
        pcall(function()
            if not self.ClientFighter or not self.ClientFighter.IsLocalPlayer then return end
            if not XC.SilentAim then return end
            local cd = r[3]
            if not cd or typeof(cd) ~= "table" then return end
            local cp = Cam.CFrame.Position
            XState.CamPos = cp
            local t, pt = xSelect({ fov=XC.SilentAimFOV, checkVis=XC.SilentAimVisCheck, partMode=XC.SilentAimTargetPart, stickyTarget=XState.SilentLastTarget, stickyBonus=XC.SilentAimStickiness or 0.05 })
            if not (t and pt) then return end
            XState.SilentLastTarget = t
            if math.random(1, 100) > (XC.SilentAimHitChance or 100) then return end
            local hp = xSP(t.Character, pt)
            if math.random(1, 100) <= (XC.SilentAimBodyMix or 0) then
                local b = t.Character:FindFirstChild("HitboxBody")
                if b and xSVis(b.Position) then hp = b end
            end
            local jit = Vector3.zero
            local jd = XC.SilentAimJitterDeg or 0
            if jd > 0 then
                local u = Vector3.new(math.random() - 0.5, math.random() - 0.5, math.random() - 0.5)
                if u.Magnitude > 1e-4 then u = u.Unit end
                local dist = (hp.Position - cp).Magnitude
                local off = math.min(math.tan(math.rad(jd)) * dist, 3) * (math.random() * 0.5)
                jit = u * off
            end
            if xEnc(cd, hp, t.Character, cp, jit) then r[4] = true end
        end)
        return unpack(r)
    end
end

    local function cal(obs, sent, g, n, seed)

        if n < CAL_FAST_N then return s, n + 1 end
        local r = math.abs(s / g)

    end

    local function step(dt)

            if last and (tick() - (XState.AimbotLastTargetTime or 0)) <= STICKY_MEM then sticky = last end
        end
        local tgt, part = acquire(cf, fovR, retainR, sticky)
        local now = tick()
        if tgt and part then _lastSeenAt = now
        else
            if _tgt and (now - _lastSeenAt) <= (XC.AimbotForgetTime or 0.2) then tgt, part = _tgt, _part
            else clearTgt(); return end
        end
        if (XC.AimbotTargetPart or "Best") == "Best" then part = resolveBest(tgt.Character, part) or part end
        if tgt ~= _prevTgt then _acquireAt = now; _ramp = 0; _haveTgt = false; _ffy, _ffp = 0, 0; _prevTgt = tgt; _flickActive = false end
        _tgt, _part = tgt, part
        XState.AimbotTarget, XState.AimbotPart = tgt, part
        XState.AimbotLastTarget, XState.AimbotLastTargetTime = tgt, now
        local targetPos = part.Position
        local tc = tgt.Character
        local trp = tc and tc:FindFirstChild("HumanoidRootPart")
        local tv = trp and trp.AssemblyLinearVelocity or Vector3.zero
        local mrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        local mv = mrp and mrp.AssemblyLinearVelocity or Vector3.zero
        local vr = tv - mv
        if XC.AimbotPrediction then
            local lead = xLead(tc, cf.Position)
            targetPos = targetPos + lead
            local hum = tc and tc:FindFirstChildOfClass("Humanoid")
            if hum and hum.FloorMaterial == Enum.Material.Air then
                local dist = (targetPos - cf.Position).Magnitude
                local lt = math.clamp(dist / 1000, 0, 0.5)
                targetPos = targetPos + Vector3.new(0, 0.5 * -196.2 * lt * lt, 0)
            end
        end
        local d = targetPos - cf.Position
        local dD = d.Magnitude
        if dD < 1e-3 then return end
        local u = d / dD
        local tYaw = yawOf(u)
        local tPit = math.clamp(pitchOf(u), -PITCH_LIMIT, PITCH_LIMIT)
        local errYaw = wrapPi(tYaw - curYaw)
        local errPit = tPit - curPit
        _errDeg = math.sqrt(errYaw * errYaw + errPit * errPit) * R2D
        local dH2 = d.X * d.X + d.Z * d.Z
        if _haveTgt and part == _lastPartRef and dH2 > 1e-4 then
            local vDotU = vr:Dot(u)
            local dH = math.sqrt(dH2)
            local rFFY = (d.Z * vr.X - d.X * vr.Z) / dH2
            local rFFP = (vr.Y - vDotU * u.Y) / dH
            local k = math.clamp(dt / FF_TAU, 0, 1)
            _ffy = _ffy + (math.clamp(rFFY, -FF_MAX, FF_MAX) - _ffy) * k
            _ffp = _ffp + (math.clamp(rFFP, -FF_MAX, FF_MAX) - _ffp) * k
        else _ffy, _ffp = 0, 0 end
        _haveTgt, _lastPartRef = true, part
        if (XC.AimbotReactionMs or 0) > 0 and (now - _acquireAt) * 1000 < XC.AimbotReactionMs then return end
        if (XC.AimbotDeadzoneDeg or 0) > 0 and math.sqrt(errYaw * errYaw + errPit * errPit) < XC.AimbotDeadzoneDeg * D2R then return end
        local sX = XC.AimbotLinkAxes and (XC.AimbotSmoothness or 0) or (XC.AimbotSmoothnessX or XC.AimbotSmoothness or 0)
        local sY = XC.AimbotLinkAxes and (XC.AimbotSmoothness or 0) or (XC.AimbotSmoothnessY or XC.AimbotSmoothness or 0)
        sX, sY = math.clamp(sX, 0, 100), math.clamp(sY, 0, 100)
        local hard = (sX == 0 and sY == 0)
        local thum = tc and tc:FindFirstChildOfClass("Humanoid")
        local air = (thum and thum.FloorMaterial == Enum.Material.Air) or math.abs(tv.Y) > 5
        if air and (XC.AimbotJumpDamping or 0) > 0 and sY > 0 then
            local jd = XC.AimbotJumpDamping / 100
            local vf = math.clamp(math.abs(tv.Y) / 60, 0, 1)
            sY = math.clamp(sY + (100 - sY) * vf * jd, 0, 100)
        end
        local aE = math.sqrt(errYaw * errYaw + errPit * errPit)
        _errEma = _errEma + (aE - _errEma) * math.clamp(dt / 0.15, 0, 1)
        _drive = _drive + dt
        local sPos = (errYaw >= 0)
        if _lastSign ~= nil and sPos ~= _lastSign then _flips = _flips + 1 end
        _lastSign = sPos
        _flips = _flips * math.exp(-dt / 0.25)
        if _flips >= 4 and _errEma > OSC_ERR and not hard then _auth = math.max(0.15, _auth * 0.5); _flips = 0
        else _auth = math.min(1, _auth + dt * 0.3) end
        local avg = (sX + sY) * 0.5
        if _errEma > RUNAWAY_ERR and _drive > (0.15 + (avg / 100) * MAX_TAU * 3) and not hard then
            _auth = math.max(0.08, _auth * 0.5)
            _gx, _gy = _sdx, _sdy; _nx, _ny = 0, 0; _drive = 0
            _trips = _trips + 1
            if _trips >= 2 then _calOff = true end
        end
        local aX = 1
        if sX > 0 then aX = (1 - math.exp(-dt / ((sX / 100) * MAX_TAU))) * _auth end
        local aY = 1
        if sY > 0 then aY = (1 - math.exp(-dt / ((sY / 100) * MAX_TAU))) * _auth end
        local cy = errYaw * aX
        local cp = errPit * aY
        if XC.AimbotCurvedFlick and _errDeg > 3.0 and not hard then
            if not _flickActive then
                _flickActive = true; _flickT = 0
                _flickDur = math.clamp((_errDeg / 90) * 0.22 + 0.08, 0.08, 0.35)
                local int = (XC.AimbotCurvedIntensity or 0.35) * (_errDeg * D2R)
                local nY, nP = -errPit / (aE + 1e-4), errYaw / (aE + 1e-4)
                local cs = ((math.floor(now * 10) % 2 == 0) and 1 or -1)
                _flick1Y, _flick1P = nY * int * cs, nP * int * cs
                _flick2Y, _flick2P = nY * int * 0.7 * cs, nP * int * 0.7 * cs
            end
            _flickT = math.min(_flickDur, _flickT + dt)
            local pr = _flickT / _flickDur
            local sj = minJerk(pr)
            cy = bezier(0, errYaw * 0.33 + _flick1Y, errYaw * 0.66 + _flick2Y, errYaw, sj) * aX
            cp = bezier(0, errPit * 0.33 + _flick1P, errPit * 0.66 + _flick2P, errPit, sj) * aY
            if pr >= 1.0 or _errDeg < 1.0 then _flickActive = false end
        else _flickActive = false end
        XState.AimbotFlickActive = _flickActive
        if not hard then local sP, sYw = springOff(); cy = cy - (sYw * aX); cp = cp - (sP * aY) end
        local os = XC.AimbotOvershoot or 0
        if os > 0 and not hard then
            _ramp = math.min(1, _ramp + dt / RAMP_TIME)
            local k = 1 + math.sin(_ramp * math.pi) * os
            cy, cp = cy * k, cp * k
        end
        local ff = (XC.AimbotTrackAssist or 100) / 100
        if ff > 0 and not hard then cy = cy + _ffy * dt * ff; cp = cp + _ffp * dt * ff end
        local nz = XC.AimbotNoiseDeg or 0
        if nz > 0 then
            local n = nz * D2R * dt
            cy = cy + (math.random() - 0.5) * 2 * n
            cp = cp + (math.random() - 0.5) * 2 * n
        end
        local cap = XC.AimbotMaxSpeed or 0
        if cap > 0 then
            local m = math.sqrt(cy * cy + cp * cp)
            local lim = cap * D2R * dt
            if m > lim then local k = lim / m; cy, cp = cy * k, cp * k end
        end
        cp = math.clamp(curPit + cp, -PITCH_LIMIT, PITCH_LIMIT) - curPit
        if _gx == 0 or _gy == 0 then return end
        local ux, uy = cy / _gx, cp / _gy
        local um = math.max(math.abs(ux), math.abs(uy))
        if um > UNIT_MAX then local k = UNIT_MAX / um; ux, uy = ux * k, uy * k end
        if _path == "mouse" then ux, _fx = quant(ux, _fx); uy, _fy = quant(uy, _fy)
        else _fx, _fy = 0, 0 end
        if ux == 0 and uy == 0 then return end
        emit(ux, uy)
        _sx, _sy = ux, uy
    end
    function XAimbot.enable()
        XC.Aimbot = true
        if _bound then return end
        R:BindToRenderStep("K4X_Aimbot", Enum.RenderPriority.Camera.Value + 1, function(dt) pcall(step, dt) end)
        _bound = true
    end
    function XAimbot.disable()
        XC.Aimbot = false; XState.AimbotKeyHeld = false
        clearTgt()
        _sx, _sy, _fx, _fy = 0, 0, 0, 0
        _haveCam, _ramp = false, 0
        _auth = 1.0
        if _bound then pcall(function() R:UnbindFromRenderStep("K4X_Aimbot") end); _bound = false end
        if _fovC then _fovC.Visible = false end
        if _lockC then _lockC.Visible = false end
        if _dbgT then _dbgT.Visible = false end
    end
end)()
